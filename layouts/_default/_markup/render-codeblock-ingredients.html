{{- /* Render hook for ```ingredients fenced blocks. Format (first line optional servings, then one ingredient per line): ```ingredients servings: 4 1.5 | lb | pork shoulder 60 | ml | vegetable oil | | salt to taste ``` Each ingredient line is `amount | unit | item`. When amount OR unit is empty the line is non-scalable (rendered verbatim, never scaled/converted). This hook emits the visible list plus the Serves/Metric-Imperial controls; the client-side scaler (static/recipe-scaler.js) does all the maths using data/units.yaml, which we serialise to JSON once per page. No numbers are computed here. */ -}} {{- $servings := 1 -}} {{- $items := slice -}} {{- range (split (trim .Inner "\n") "\n") -}} {{- $line := trim . " \t\r" -}} {{- if $line -}} {{- if strings.HasPrefix (lower $line) "servings:" -}} {{- $servings = int (trim (index (split $line ":") 1) " ") -}} {{- else if strings.HasPrefix $line "#" -}} {{- /* a group subheading, e.g. `# Sauce` */ -}} {{- $items = $items | append (dict "group" true "item" (trim $line "# ")) -}} {{- else -}} {{- $parts := split $line "|" -}} {{- if ge (len $parts) 3 -}} {{- $items = $items | append (dict "amount" (trim (index $parts 0) " ") "unit" (trim (index $parts 1) " ") "item" (trim (delimit (after 2 $parts) "|") " ")) -}} {{- else -}} {{- $items = $items | append (dict "amount" "" "unit" "" "item" $line) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if not (.Page.Scratch.Get "recipeScalerUnits") -}} {{- .Page.Scratch.Set "recipeScalerUnits" true -}} {{- end -}}
Serves
    {{- range $items -}} {{- if .group -}}
  • {{ .item }}
    • {{- else if and .amount .unit -}}
  • {{ .amount }} {{ .unit }} {{ .item }}
    • {{- else -}}
  • {{ .item }}
    • {{- end -}} {{- end -}}
